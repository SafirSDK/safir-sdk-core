From .github/actions/wireguard-overlay/action.yml, the 'Punch and verify
the tunnel (Linux)' step - this is ALL the connectivity verification that
exists before the multicomputer test suite launches:

    - name: Punch and verify the tunnel (Linux)
      if: ${{ runner.os == 'Linux' && inputs.verify-peer-ip != '' }}
      shell: bash
      env:
        VERIFY_PEER_IP: ${{ inputs.verify-peer-ip }}
      run: |
        set -uo pipefail
        # Pinging the peer's wg address drives the NAT punch (both peers do this
        # concurrently). We gate on the actual WireGuard HANDSHAKE, not on an ICMP
        # reply: the encrypted tunnel can be fully up and carrying traffic while
        # ICMP echo between the transit /32s still doesn't round-trip. Dob is UDP
        # anyway, so a completed handshake is the signal that matters here.
        ok=0
        for i in $(seq 1 90); do
          ping -c1 -W1 "$VERIFY_PEER_IP" >/dev/null 2>&1 || true
          hs="$(sudo wg show wg0 latest-handshakes | awk '{print $2}')"
          if [ -n "$hs" ] && [ "$hs" != "0" ]; then
            ok=1; echo "tunnel up: handshake registered (attempt $i, ts=$hs)"; break
          fi
          sleep 1
        done
        echo "=== wg show ==="; sudo wg show
        if [ "$ok" != "1" ]; then
          echo "WireGuard tunnel did not come up (no handshake with peer)"
          exit 1
